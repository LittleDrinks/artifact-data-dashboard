#### 2.2.1 三种模式实现

```typescript
// ai-mode.service.ts

enum AIMode {
  GRAPH = 'graph',      // 图谱模式 - 只查图谱
  KNOWLEDGE = 'knowledge',  // 知识模式 - 查图谱+归纳
  GENERAL = 'general',   // 通用模式 - 自由回答
}

interface ModeConfig {
  mode: AIMode;
  useMCP: boolean;      // 是否使用MCP工具
  useRetrieval: boolean; // 是否先检索上下文
  systemPrompt: string;
}

const modeConfigs: Record<AIMode, ModeConfig> = {
  [AIMode.GRAPH]: {
    mode: AIMode.GRAPH,
    useMCP: true,
    useRetrieval: false,
    systemPrompt: `你是一个文物知识图谱助手。你只能基于知识图谱中的数据回答问题。
如果图谱中没有相关信息，请明确告知用户。
回答要准确、简洁，并标注数据来源。`,
  },
  
  [AIMode.KNOWLEDGE]: {
    mode: AIMode.KNOWLEDGE,
    useMCP: true,
    useRetrieval: true,
    systemPrompt: `你是一个文物研究助手。你会先查询知识图谱获取实例数据，
然后基于这些实例归纳知识，回答用户的问题。
回答要有学术深度，并引用具体数据支撑。`,
  },
  
  [AIMode.GENERAL]: {
    mode: AIMode.GENERAL,
    useMCP: false,
    useRetrieval: false,
    systemPrompt: `你是一个文物领域的AI助手。你可以基于你的训练数据自由回答用户问题。
如果用户问题涉及具体文物，建议切换到图谱模式或知识模式以获得更准确的信息。`,
  },
};

// 模式切换流程
async function handleChat(message: string, mode: AIMode, context: ChatContext) {
  const config = modeConfigs[mode];
  
  let tools = [];
  let retrievalContext = '';
  
  // 1. 准备工具
  if (config.useMCP) {
    tools = await getAvailableTools();
  }
  
  // 2. 预检索（知识模式）
  if (config.useRetrieval) {
    const relevantEntities = await searchEntities(message);
    retrievalContext = formatEntitiesContext(relevantEntities);
  }
  
  // 3. 构建提示
  const prompt = buildPrompt(message, config, retrievalContext);
  
  // 4. 调用AI
  const response = await callAI({
    prompt,
    tools: config.useMCP ? tools : undefined,
    stream: true,
  });
  
  return response;
}
```

  coreLayer: GraphNode[];

  detailLayer: GraphNode[];

    if (viewport.zoom < 0.5) {

      this.renderClusters();
    } else if (viewport.zoom < 1.5) {
      // 中距离：显示核心节点
      this.renderCoreLayer();
    } else {
      // 近距离：显示全部
      this.renderAllLayers();
    }
  }

class VirtualizedGraph {
  visibleNodes: Set<string> = new Set();
  
  updateVisibleNodes(viewport: Viewport) {
    const bounds = this.getViewportBounds(viewport);
    
    this.visibleNodes.clear();
    for (const node of this.allNodes) {
      if (this.isInBounds(node, bounds)) {
        this.visibleNodes.add(node.id);
      }
    }
    
    this.render();
  }
}

## 5. 安全设计

### 5.1 认证授权

```typescript
// JWT认证流程

// 1. 登录获取Token
POST /api/auth/login
{
  "username": "user@example.com",
  "password": "********"
}

Response:
{
  "accessToken": "eyJhbGciOiJIUzI1NiIs...",
  "refreshToken": "eyJhbGciOiJIUzI1NiIs...",
  "expiresIn": 3600
}

// 2. 请求携带Token
Headers:
Authorization: Bearer {accessToken}

// 3. Token验证中间件
const authMiddleware = (req, res, next) => {
  const token = req.headers.authorization?.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ code: 'E0002', message: '未提供认证令牌' });
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded;
    next();
  } catch (err) {
    return res.status(401).json({ code: 'E0002', message: '令牌无效或已过期' });
  }
};
```

### 5.2 Cypher查询安全

```typescript
// Cypher查询黑名单
const CYPHER_BLACKLIST = [
  // 破坏性操作
  'DELETE', 'DETACH DELETE', 'REMOVE',
  'CREATE', 'DROP', 'SET',
  
  // 系统操作
  'CALL dbms', 'CALL db',
  'apoc', 'gds',
  
  // 文件操作
  'LOAD CSV', 'LOAD DATA',
];

// 查询验证中间件
const validateCypher = (query: string): boolean => {
  const upperQuery = query.toUpperCase();
  
  for (const keyword of CYPHER_BLACKLIST) {
    if (upperQuery.includes(keyword)) {
      return false;
    }
  }
  
  // 只允许MATCH查询
  if (!upperQuery.trim().startsWith('MATCH')) {
    return false;
  }
  
  return true;
};

// 执行限制
const executeCypher = async (query: string, parameters?: any) => {
  // 1. 验证查询
  if (!validateCypher(query)) {
    throw new Error('非法Cypher查询');
  }
  
  // 2. 添加执行限制
  const session = neo4jDriver.session({
    defaultAccessMode: neo4j.session.READ,  // 只读模式
    database: 'neo4j',
  });
  
  // 3. 设置超时
  const result = await session.run(query, parameters, {
    timeout: 5000,  // 5秒超时
  });
  
  await session.close();
  return result;
};
```
